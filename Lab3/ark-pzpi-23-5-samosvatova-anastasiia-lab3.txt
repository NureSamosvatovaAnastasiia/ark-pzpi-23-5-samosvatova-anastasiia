3.2 Інтелектуальна автоматизація мікроклімату
     Для забезпечення оптимальних умов вирощування було розроблено сервіс автоматизації AutomationService (див. дод. В), який працює за принципом зворотного зв'язку. Система не просто реагує на порогові значення, а використовує адаптивні алгоритми.
     Для керування освітленням реалізовано алгоритм компенсації сонячного світла за формулою:
L artificial = L target – L natural
де:
• L target – цільовий рівень освітленості для культури;
• L natural– поточний рівень природного світла від сенсора.
   
     При цьому враховуються циркадні ритми рослин: якщо поточний час виходить за межі світлового дня (наприклад, 20:00–06:00), система примусово вимикає освітлення для забезпечення періоду спокою. Для поливу реалізовано розрахунок необхідного об’єму води на основі дефіциту вологості та площі теплиці:
V water = H target – H current * S area* K crop
     де K crop – коефіцієнт вологоємності культури. 
     Для терморегуляції розраховується необхідна теплова потужність для нагріву об'єму повітря теплиці за заданий час з урахуванням тепловтрат
     Для підвищення стабільності системи та запобігання «дрібезгу» виконавчих механізмів (частих вмикань/вимикань) було впроваджено аналіз трендів зміни показників (рядок 1-22).

У такому випадку система блокує повторне ввімкнення, запобігаючи занадто стрімкого росту тренду.

226.                   const fanCapacityM3H = parseFloat(fan.capacity) || 5000;
227.       
228.                   let requiredFlow = volume * (10 + deltaT * 10); 
229.     
230. 
231.                   if (trend > 0) requiredFlow += (volume * 200 * trend); 
232.               
233.                   else requiredFlow += (volume * 500 * trend); 
234.     
235.                   let percent = (requiredFlow / fanCapacityM3H) * 100;
236.                   percent = Math.min(100, Math.max(0, percent));
237.     
238.                   if (percent > 10) await this.setActuator(fan, true, percent, `AUTO:Cool(Target ${target})`);
239.                   else await this.setActuator(fan, false, 0, 'AUTO:Coast');
240.               } else {
241.                   await this.setActuator(fan, false, 0, 'AUTO:ComfortZone');
242.               }
243.           }
244.           
245.           if (vent) {
246.                if (predictedVal > max || currentVal > max) {
247.                    let angle = (deltaT * 15) + (trend * 500);
248.                    angle = Math.min(100, Math.max(0, angle));
249.                    if (angle > 5) await this.setActuator(vent, true, angle, `AUTO:Vent`);
250.                    else await this.setActuator(vent, false, 0, 'AUTO:Close');
251.                } else {
252.                    await this.setActuator(vent, false, 0, 'AUTO:ComfortZone');
253.                }
254.           }
255.           if (heater) await this.setActuator(heater, false, 0, 'AUTO:StopHeat');
256.       }
257.       else {
258.           if (heater) await this.setActuator(heater, false, 0, 'AUTO:Optimal');
259.           if (fan) await this.setActuator(fan, false, 0, 'AUTO:Optimal');
260.           if (vent) await this.setActuator(vent, false, 0, 'AUTO:Optimal');
261.       }
262.   }

265.   async handleAirHumidity(greenhouse, crop, currentVal, trend) {
266.       const min = parseFloat(crop.idealAirHumidityMin);
267.       const max = parseFloat(crop.idealAirHumidityMax);
268.       const keyPhrase = 'Humidity';
269. 
270.       const predictedVal = currentVal + (trend * PREDICTION_TIME_SECONDS);
271.       const target = (min + max) / 2;
272.       const humidifier = await this.findActuator(greenhouse.id, 'humidifier');
273.       const vent = await this.findActuator(greenhouse.id, 'vent');
274. 
275.       if (currentVal > (max + 5)) {
276.            await this.checkAndAlert(greenhouse.id, `Warning: High humidity (${currentVal}%). Risk of mold!`, 'WARNING', keyPhrase);
277.       } else if (currentVal < (min - 5)) {
278.            await this.checkAndAlert(greenhouse.id, `Warning: Low humidity (${currentVal}%). Risk of drying!`, 'WARNING', keyPhrase);
279.       } else if (currentVal >= min && currentVal <= max) {
280.            await this.checkAndAlertRecovery(greenhouse.id, `Humidity normalized.`, keyPhrase);
281.       }
282.       if (predictedVal < min) {
283.           if (humidifier) {
284.         
285.               const maxWaterGM3 = this.calculateMaxWaterInAir(currentTemp);
286. 
287.               const deficitPercent = target - currentVal; 
288.               const deficitGM3 = maxWaterGM3 * (deficitPercent / 100);
289. 
290.               const volume = parseFloat(greenhouse.areaSqMeters) * parseFloat(greenhouse.heightMeters || 3);
291.               const totalWaterNeededMl = deficitGM3 * volume; 
292. 
293.               const capacityMlH = parseFloat(humidifier.capacity) || 500; 
294.               
295.               let power = (totalWaterNeededMl / capacityMlH) * 100 * 2; 
296.           
297.               power -= (trend * 1000); 
298. 
299.               power = Math.min(100, Math.max(10, power));
300. 
301.               await this.setActuator(humidifier, true, power, `AUTO:Humidify(Need ${totalWaterNeededMl.toFixed(0)}ml)`);
302.           }
303.       } 
304. 
305.       else if (predictedVal > max) {
306.           if (vent) {
307.               const excess = currentVal - target;
308.               let power = (excess * 5) + (trend * 1000);
309.               power = Math.min(100, Math.max(10, power));
310.               
311.              await this.setActuator(vent, true, power, `AUTO:Dehumidify(Target ${target})`);
312.           }
313.           if (humidifier) await this.setActuator(humidifier, false, 0, 'AUTO:StopHumidify');
314.       }
315.       else {
316.           if (humidifier) await this.setActuator(humidifier, false, 0, 'AUTO:Optimal');
317.       }
318.   }

321.   async handleLighting(greenhouse, crop, currentLux, currentTime = new Date()) {
322.       const targetLux = crop.idealLightLevel;
323.       const dayLength = crop.requiredDayHours;
324.       const area = parseFloat(greenhouse.areaSqMeters);
325.       const keyPhrase = 'Light';
326.     
327.      const currentHour = currentTime.getHours();
328.       const isDayTime = currentHour >= 6 && currentHour < (6 + dayLength);
329. 
330.       if (isDayTime) {
331.           if (currentLux  < (targetLux * 0.5)) {
332.               await this.checkAndAlert(greenhouse.id, `CRITICAL: Low light (${currentLux} lux). Growth stalled!`, 'CRITICAL', keyPhrase);
333.           } else if (currentLux  < (targetLux * 0.8)) {
334.               await this.checkAndAlert(greenhouse.id, `Warning: Low light level (${currentLux} lux).`, 'WARNING', keyPhrase);
335.           } else if (currentLux  >= targetLux) {
336.               await this.checkAndAlertRecovery(greenhouse.id, `Light level optimal.`, keyPhrase);
337.           }
338.       }
339.       const growLight = await this.findActuator(greenhouse.id, 'grow_light');
340.       if (!growLight) return;
341. 
342.       if (isDayTime) {
343.          
344.           const maxLampLux = (parseFloat(growLight.capacity) || 20000) / area;
345.           const currentLampContribution = growLight.currentState 
346.               ? (maxLampLux * (parseFloat(growLight.currentValue) / 100)) 
347.               : 0;
348. 
349.           const naturalLux = Math.max(0, currentLux - currentLampContribution);
350.           
351.           if (naturalLux < targetLux) {
352.               const neededArtificial = targetLux - naturalLux;
353.               let percent = (neededArtificial / maxLampLux) * 100;
354.               percent = Math.min(100, Math.max(0, percent));
355. 
356.               await this.setActuator(growLight, true, percent, `AUTO:SmartLight`);
357.           } else {
358.               await this.setActuator(growLight, false, 0, 'AUTO:SunEnough');
359.           }
360.       } else {
361.           await this.setActuator(growLight, false, 0, 'AUTO:NightMode');
362.       }
363.   }

388.       const currentVal = parseFloat(actuator.currentValue || 0);
389.       const diff = Math.abs(currentVal - value);
390.     
391.       if (actuator.currentState !== state || (state === true && diff > 5)) {
392.           console.log(` [SMART-SET] ${actuator.type}: ${value.toFixed(1)}% (was ${currentVal.toFixed(1)}%)`);
393.           await IoTRepo.updateActuatorState(actuator.id, state, value, reason);

Ці маршрути призначені для глобального керування системою, моніторингу та управління користувачами.

   Отримання списку всіх зареєстрованих користувачів.

Опис: Видалення користувача із системи.

   Зміна ролі користувача (наприклад, підвищення до admin або пониження до user).

   Перегляд глобальних логів системи (останні 100 дій автоматики або користувачів у всій системі).
6. GET /admin/greenhouses
   Отримання списку всіх теплиць у системі (незалежно від власника) для моніторингу.
